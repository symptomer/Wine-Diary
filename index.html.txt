        function switchTab(tabName) {
            document.querySelectorAll('.container').forEach(c => c.classList.remove('active'));
            document.querySelectorAll('.tab-item').forEach(t => t.classList.remove('active'));
            
            document.getElementById(`tab-${tabName}`).classList.add('active');
            event.currentTarget.classList.add('active');

        function previewImage(input, targetId) {
            if (input.files && input.files[0]) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    const img = document.getElementById(targetId);
                    img.src = e.target.result;
                    img.style.display = 'block';
                }
                reader.readAsDataURL(input.files[0]);
            }
        }

        // 구글 CORS 제약을 우회하는 가장 깔끔한 인라인 이미지 분석 로직
        async function processBackLabel(input) {
            previewImage(input, 'imgBack');
            if (!input.files || !input.files[0]) return;

            const statusDiv = document.getElementById('aiStatus');
            statusDiv.style.display = 'block';

            const file = input.files[0];
            const reader = new FileReader();
            
            reader.onload = async function(e) {
                const base64Data = e.target.result.split(',')[1];
                
                // 브라우저에서 직접 요청 가능한 정식 v1beta 주소와 페이로드 규격 설정
                const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=${GEMINI_API_KEY}`;
                
                const prompt = "This is a wine back label. Extract the following information in strict JSON format: {name: 'Wine Name', vintage: 'Year or null', region: 'Country and Region', variety: 'Grape variety', abv: 'Alcohol % without percentage sign'}. Answer ONLY with the raw JSON object, do not wrap in markdown block.";

                try {
                    const response = await fetch(url, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({
                            contents: [{
                                parts: [
                                    { text: prompt },
                                    { inlineData: { mimeType: file.type, data: base64Data } }
                                ]
                            }]
                        })
                    });

                    const result = await response.json();
                    const txt = result.candidates[0].content.parts[0].text.trim();
                    
                    // 마크다운 흔적 방어 코드
                    const cleanJson = txt.replace(/```json|```/g, "");
                    const data = JSON.parse(cleanJson);

                    if(data.name) document.getElementById('wName').value = data.name;
                    if(data.vintage) document.getElementById('wVintage').value = data.vintage;
                    if(data.region) document.getElementById('wRegion').value = data.region;
                    if(data.variety) document.getElementById('wVariety').value = data.variety;
                    if(data.abv) document.getElementById('wAbv').value = data.abv;

                    statusDiv.innerText = "🎉 AI 라벨 분석 완료!";
                    setTimeout(() => statusDiv.style.display = 'none', 2000);

                } catch (err) {
                    console.error(err);
                    statusDiv.innerText = "❌ AI 분석 실패 (텍스트로 직접 보완 가능)";
                }
            };
            reader.readAsDataURL(file);
        }

        function initStars() {
            document.querySelectorAll('.star-rating').forEach(container => {
                const taste = container.getAttribute('data-taste');
                const stars = container.querySelectorAll('.star');
                stars.forEach(star => {
                    star.addEventListener('click', () => {
                        const val = parseInt(star.getAttribute('data-value'));
                        ratings[taste] = val;
                        stars.forEach((s, idx) => {
                            if(idx < val) s.classList.add('selected');
                            else s.classList.remove('selected');
                        });
                    });
                });
            });
        }

        function renderCellar() {
            const grid = document.getElementById('cellarGrid');
            grid.innerHTML = '';
            for (let i = 1; i <= 10; i++) {
                const wine = wineStorage[i];
                let html = `<div class="cellar-slot ${wine ? 'occupied' : ''}" onclick="clickSlot(${i})" id="slot-${i}">`;
                html += `<span class="slot-num">${i}번 칸</span>`;
                if(wine) {
                    if(wine.frontImg) html += `<div style="position:absolute; inset:0; background-image:url(${wine.frontImg}); background-size:cover; background-position:center; border-radius:10px; z-index:1;"></div>`;
                    html += `<div class="slot-info" style="z-index:2;">${wine.name}</div>`;
                } else {
                    html += `<span style="font-size:24px; color:#444;">+</span><span style="font-size:12px; color:var(--text-muted);">빈 칸</span>`;
                }
                html += `</div>`;
                grid.innerHTML += html;
            }
        }

        function clickSlot(num) {
            if(wineStorage[num]) {
                openModal(wineStorage[num]);
            } else {
                document.getElementById('selectedSlot').value = num;
                switchTab('record');
                alert(`${num}번 칸에 보관할 와인 정보를 작성해 주세요.`);
            }
        }

        function saveWine() {
            const name = document.getElementById('wName').value;
            let slot = document.getElementById('selectedSlot').value;
            
            if(!name) return alert('와인 이름을 입력해 주세요.');
            if(!slot) {
                // 빈 칸 탐색 정렬 알고리즘
                for(let i=1; i<=10; i++) {
                    if(!wineStorage[i]) { slot = i; break; }
                }
                if(!slot) return alert('현재 와인셀러 10칸이 꽉 찼습니다! 기존 와인을 비우거나 상세페이지에서 수동 관리해 주세요.');
            }

            const wineData = {
                id: Date.now(),
                slot: slot,
                name: name,
                vintage: document.getElementById('wVintage').value,
                region: document.getElementById('wRegion').value,
                variety: document.getElementById('wVariety').value,
                price: document.getElementById('wPrice').value,
                abv: document.getElementById('wAbv').value,
                ratings: { ...ratings },
                memo: document.getElementById('wMemo').value,
                frontImg: document.getElementById('imgFront').src || null
            };

            wineStorage[slot] = wineData;
            localStorage.setItem('symptomer_wines', JSON.stringify(wineStorage));
            
            // 폼 초기화
            resetForm();
            renderCellar();
            renderList();
            switchTab('cellar');
        }
